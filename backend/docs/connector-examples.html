  2. helper 的「剪映草稿根目录」已配置（否则草稿创建会失败，code 1001）

    共享底层工具函数

    所有组件依赖以下三个函数（discoverHelper / fetchHealth / streamTask）。 放到你的项目里某个 utils 文件，按需 import。

    const HELPER_PORT_KEY = 'capcut_helper_port'
const HELPER_PORT_RANGE = [9527, 9528, 9529, 9530, 9531, 9532, 9533, 9534, 9535, 9536]

async function discoverHelper({ timeoutMs = 500 } = {}) {
  const cached = Number(localStorage.getItem(HELPER_PORT_KEY)) || null
  const candidates = cached
    ? [cached, ...HELPER_PORT_RANGE.filter((p) => p !== cached)]
    : HELPER_PORT_RANGE
  for (const port of candidates) {
    try {
      const resp = await fetch(`http://localhost:${port}/api/v1/health`, {
        signal: AbortSignal.timeout(timeoutMs),
      })
      if (!resp.ok) continue
      const body = await resp.json()
      const health = body?.data
      if (health?.service === 'capcut_helper') {
        localStorage.setItem(HELPER_PORT_KEY, String(port))
        return { port, health }
      }
    } catch { /* 继续下一个 */ }
  }
  return null
}

async function fetchHealth(port) {
  const resp = await fetch(`http://localhost:${port}/api/v1/health`)
  return (await resp.json()).data
}

/**
 * 订阅 SSE stream_url 拿任务实时进度。
 * onProgress(task) 每次状态/字节进度变化触发；onDone(task) 任务结束（done/failed）触发；
 * 返回一个 close 函数，调用方主动取消时用。
 */
function streamTask(port, streamUrl, { onProgress, onDone, onError } = {}) {
  const es = new EventSource(`http://localhost:${port}${streamUrl}`)
  es.addEventListener('progress', (e) => onProgress?.(JSON.parse(e.data)))
  es.addEventListener('done', (e) => {
    es.close()
    onDone?.(JSON.parse(e.data))
  })
  es.onerror = (e) => { onError?.(e) }
  return () => es.close()
}
