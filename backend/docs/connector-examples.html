capcut_helper 连接组件示例

给调用方应用接入剪映助手的 4 套现成 UI 组件 — 极简徽章、状态卡片、内嵌按钮、完整向导。
全部纯 HTML + vanilla JS，复制即用，无任何依赖。

全局状态 未探测

1. 极简徽章

Minimal

小圆点 + 状态文字，浮在页面角落不抢戏。未授权时整个徽章可点击直接唤起授权。

查看代码 
<!-- HTML -->
<div id="capcutBadge" class="ch-badge" data-state="scan">
  <span class="dot"></span><span class="text">探测中</span>
</div>

<!-- CSS（按需复制顶部 .ch-badge 段） -->

<script>
async function mountBadge() {
  const el = document.getElementById('capcutBadge')
  const text = el.querySelector('.text')
  const result = await discoverHelper()           // 见文末「共享底层」
  if (!result) {
    el.dataset.state = 'err'
    text.textContent = '剪映助手未运行'
    el.classList.remove('clickable')
    el.onclick = null
    return
  }
  const { health } = result
  if (health.cors_allowed === false) {
    el.dataset.state = 'warn'
    text.textContent = '需要授权'
    el.classList.add('clickable')
    el.onclick = () => { location.href = health.trust_url }
  } else {
    el.dataset.state = 'ok'
    text.textContent = '已连接 v' + health.version
    el.classList.remove('clickable')
    el.onclick = null
  }
}
mountBadge()
</script>

2. 状态卡片

Card

完整信息展示 + 上下文 action 按钮。适合放在调用方设置页或集成入口。

查看代码 
<!-- HTML -->
<div id="capcutCard" class="ch-card">
  <div class="head">
    <span class="title">剪映助手</span>
    <span class="pill" data-state="scan">探测中</span>
  </div>
  <div class="body">正在扫描本地端口...</div>
  <div class="actions"></div>
</div>

<script>
async function mountCard() {
  const root = document.getElementById('capcutCard')
  const pill = root.querySelector('.pill')
  const body = root.querySelector('.body')
  const actions = root.querySelector('.actions')

  const result = await discoverHelper()
  if (!result) {
    pill.dataset.state = 'err'
    pill.textContent = '离线'
    body.innerHTML = '未在端口 9527–9536 找到剪映助手。请先启动桌面应用。'
    actions.innerHTML = `<button class="btn" onclick="mountCard()">重新探测</button>`
    return
  }

  const { health } = result
  if (health.cors_allowed === false) {
    pill.dataset.state = 'warn'
    pill.textContent = '需要授权'
    body.innerHTML = `当前页面 <code>${health.your_origin}</code> 还未被剪映助手信任，`
                   + `点下方按钮唤起授权弹窗。`
    actions.innerHTML = `
      <button class="btn primary" onclick="location.href='${health.trust_url}'">允许接入</button>
      <button class="btn" onclick="mountCard()">我已授权，重试</button>`
  } else {
    pill.dataset.state = 'ok'
    pill.textContent = '已连接'
    body.innerHTML = `<span class="strong">v${health.version}</span> · 端口 ${health.port}`
    actions.innerHTML = ''
  }

  if (health.has_update) {
    const hint = document.createElement('div')
    hint.className = 'update-hint'
    hint.innerHTML = `🎉 新版 v${health.latest_version} 可用 ·
      <a onclick="window.open('${health.release_url}','_blank')">查看更新</a>`
    root.appendChild(hint)
  }
}
mountCard()
</script>

3. 内嵌按钮

Inline

单按钮形态，状态变化驱动文案、颜色、点击行为切换。适合塞进调用方应用的工具栏。

查看代码 
<!-- HTML -->
<button id="capcutInline" class="ch-inline" data-state="scan">
  <span class="icon">⏳</span><span class="text">探测中</span>
</button>

<script>
async function mountInline() {
  const btn = document.getElementById('capcutInline')
  const icon = btn.querySelector('.icon')
  const text = btn.querySelector('.text')
  btn.onclick = null

  const result = await discoverHelper()
  if (!result) {
    btn.dataset.state = 'err'
    icon.textContent = '⚠'
    text.textContent = '剪映助手未运行 · 点击重试'
    btn.onclick = mountInline
    return
  }
  const { health } = result
  if (health.cors_allowed === false) {
    btn.dataset.state = 'warn'
    icon.textContent = '🔓'
    text.textContent = '点击授权剪映助手'
    btn.onclick = () => { location.href = health.trust_url }
  } else {
    btn.dataset.state = 'ok'
    icon.textContent = '✓'
    text.textContent = `剪映助手已连接 v${health.version}`
    btn.onclick = null
  }
}
mountInline()
</script>

4. 完整向导

Wizard

分步骤展示「发现 → 自检 → 授权 → 测试」全流程。适合首次接入引导页 / 故障诊断面板。

查看代码 
<!-- HTML -->
<div id="capcutWizard" class="ch-wizard">
  <h3>接入剪映助手</h3>
  <p class="sub">按以下步骤完成连接，之后即可一键导出到剪映草稿。</p>
  <ol class="steps">
    <li data-status="active" data-step="discover">
      <span class="step-num">1</span>
      <div class="step-body">
        <div class="step-title">发现本地服务</div>
        <div class="step-meta">扫描端口 9527–9536</div>
      </div>
    </li>
    <li data-status="pending" data-step="cors">
      <span class="step-num">2</span>
      <div class="step-body">
        <div class="step-title">检查跨域授权</div>
        <div class="step-meta">确认本页面在白名单</div>
      </div>
    </li>
    <li data-status="pending" data-step="done">
      <span class="step-num">3</span>
      <div class="step-body">
        <div class="step-title">就绪</div>
        <div class="step-meta">可以调用业务接口</div>
      </div>
    </li>
  </ol>
  <div class="wizard-action"></div>
</div>

<script>
async function runWizard() {
  const root = document.getElementById('capcutWizard')
  const setStep = (key, status, meta) => {
    const li = root.querySelector(`li[data-step="${key}"]`)
    li.dataset.status = status
    if (meta) li.querySelector('.step-meta').innerHTML = meta
  }
  const setAction = (html) => { root.querySelector('.wizard-action').innerHTML = html }

  setStep('discover', 'active'); setStep('cors', 'pending'); setStep('done', 'pending'); setAction('')

  const result = await discoverHelper()
  if (!result) {
    setStep('discover', 'failed', '端口段全部无响应')
    setAction(`<button class="btn primary" onclick="runWizard()">重新探测</button>`)
    return
  }
  setStep('discover', 'done', `端口 ${result.port} · v${result.health.version}`)

  setStep('cors', 'active')
  if (result.health.cors_allowed === false) {
    setStep('cors', 'failed', `当前 origin <code>${result.health.your_origin}</code> 不在白名单`)
    setAction(`
      <button class="btn primary" onclick="location.href='${result.health.trust_url}'">允许接入</button>
      <button class="btn" onclick="runWizard()">我已授权，重试</button>`)
    return
  }
  setStep('cors', 'done', '已授权')
  setStep('done', 'done', '可以调用 /api/v1/drafts')
  setAction('')
}
runWizard()
</script>

共享底层工具函数

Shared

上面所有组件都依赖这两个函数（discoverHelper / requestTrust）。 放在你的项目里某个 utils 文件，按需 import。

没有可视化预览 — 见下方代码。
查看代码 
const HELPER_PORT_KEY = 'capcut_helper_port'
const HELPER_PORT_RANGE = [9527, 9528, 9529, 9530, 9531, 9532, 9533, 9534, 9535, 9536]

/**
 * 扫端口段找 capcut_helper 本地服务。命中后返回 { port, health }；全部失败返回 null。
 * 顺序：先尝试 localStorage 缓存的端口（如果有），失败再扫整段。
 */
async function discoverHelper({ timeoutMs = 500 } = {}) {
  const cached = Number(localStorage.getItem(HELPER_PORT_KEY)) || null
  const candidates = cached
    ? [cached, ...HELPER_PORT_RANGE.filter((p) => p !== cached)]
    : HELPER_PORT_RANGE

  for (const port of candidates) {
    try {
      const resp = await fetch(`http://localhost:${port}/api/v1/health`, {
        signal: AbortSignal.timeout(timeoutMs),
      })
      if (!resp.ok) continue
      const body = await resp.json()
      const health = body?.data
      if (health?.service === 'capcut_helper') {
        localStorage.setItem(HELPER_PORT_KEY, String(port))
        return { port, health }
      }
    } catch {
      // 端口无服务 / 超时 / 不是 capcut_helper，继续尝试下一个
    }
  }
  return null
}

/**
 * 唤起剪映助手桌面应用弹出授权确认。注意：浏览器会先弹「是否打开 capcut-helper」
 * 系统确认框，用户点确认后才会拉起应用。链接由 health 返回，origin 已正确编码。
 */
function requestTrust(health) {
  if (!health?.trust_url) {
    throw new Error('health.trust_url 不存在 — 只在 cors_allowed=false 时才会下发')
  }
  location.href = health.trust_url
}

接入要点

  • 所有组件都依赖 discoverHelper() 探测端口段 9527–9536。必须以 health 返回的 service === "capcut_helper" 为准，不要假设某个端口就是 helper（可能撞到本机别的服务）。
  • cors_allowed === false 时一定要给用户明确指引，不要等业务接口报 Failed to fetch 才反应。指引文案可以直接用 health.hint。
  • 点击 trust_url 浏览器会先弹一个系统级「是否打开 capcut-helper」确认框，用户点允许后才会拉起 helper 桌面应用并弹授权 Modal。这是浏览器安全模型，不可绕过，文案要让用户预期到这个跳转。
  • 用户在 helper 里点「允许」后，CORS 白名单是热生效的（不需要重启 helper）。所以「允许接入」按钮后面紧跟一个「我已授权，重试」按钮就足够，无需提示「请重启应用」。
  • 所有组件的视觉风格（颜色、圆角、间距）都通过页面顶部的 CSS 变量（:root）控制，按你自己的设计系统改即可。暗色模式自动适配。
  • 更详细的字段说明与边界情况见 CALLER_GUIDE.md。
